**Objective**: Simulate the motion of two bodies interacting via Newtonian gravity in 2D. Structure your code using **modules** and define a Body **type/class** that stores each particle’s state and **provides methods** for updating it.

## Physics Background

Each body is characterized by:
- Position: $(x, y)$
- Velocity: $(v_x, v_y)$
- Mass: $m$

The **force between two bodies** is given by Newton’s law of universal gravitation:

$$ \vec{F} = G \frac{m_1 m_2}{r^2} \hat{r} $$

Use a simple **Euler integration** to update positions and velocities over time:

$$ \vec{a} = \frac{\vec{F}}{m}\ ,$$ 
$$\vec{v}(t + \Delta t) = \vec{v}(t) + \vec{a} \Delta t\ ,$$
$$\vec{x}(t + \Delta t) = \vec{x}(t) + \vec{v}(t) \Delta t $$

### Requirements

1. Define a Body type/class:
- Mass
- Position (2D)
- Velocity (2D)

C: use `struct`
Fortran: use class with `type`

2. Write a **module gravity** that provides:
- `compute_force(b1, b2, fx, fy)` — computes the gravitational force on `b1` due to `b2`
- `update_body(b, fx, fy, dt)` — applies the force to update `b` using Euler’s method
- `print_body(b)` — prints the body state

4. In the **main program**:
- Define two bodies (e.g., Sun and Earth-like)
- Simulate motion for 1000 steps with `dt = 0.01`
- Print or store their trajectories

```Fortran
! planetary_motion.f90

MODULE gravity
  IMPLICIT NONE
  PRIVATE
  PUBLIC :: Body, compute_force, update_body, print_body

  REAL, PARAMETER :: G = 1.0 ! Gravitational constant (arbitrary units)

  TYPE :: Body
    REAL :: mass
    REAL :: pos(2)
    REAL :: vel(2)
  CONTAINS
    PROCEDURE :: print => print_body
  END TYPE Body

CONTAINS

  SUBROUTINE compute_force(b1, b2, fx, fy)
    TYPE(Body), INTENT(IN) :: b1, b2
    REAL, INTENT(OUT) :: fx, fy
    REAL :: dx, dy, r2, r, f

    dx = b2%pos(1) - b1%pos(1)
    dy = b2%pos(2) - b1%pos(2)
    r2 = dx**2 + dy**2
    r = SQRT(r2)

    IF (r2 == 0.0) THEN
      fx = 0.0
      fy = 0.0
    ELSE
      f = G * b1%mass * b2%mass / r2
      fx = f * dx / r
      fy = f * dy / r
    END IF
  END SUBROUTINE compute_force

  SUBROUTINE update_body(b, fx, fy, dt)
    TYPE(Body), INTENT(INOUT) :: b
    REAL, INTENT(IN) :: fx, fy, dt
    REAL :: ax, ay

    ax = fx / b%mass
    ay = fy / b%mass

    b%vel(1) = b%vel(1) + ax * dt
    b%vel(2) = b%vel(2) + ay * dt

    b%pos(1) = b%pos(1) + b%vel(1) * dt
    b%pos(2) = b%pos(2) + b%vel(2) * dt
  END SUBROUTINE update_body

  SUBROUTINE print_body(this,step,unit)
    CLASS(Body), INTENT(IN) :: this
    INTEGER, INTENT(IN) :: step
    WRITE(unit=unit,*) step, this%pos(:), this%vel(:)
  END SUBROUTINE print_body

END MODULE gravity
```

```Fortran
PROGRAM two_body_sim
  USE gravity
  IMPLICIT NONE

  TYPE(Body) :: sun, earth
  REAL :: fx, fy
  REAL :: dt
  INTEGER :: i, nsteps
  INTEGER :: unit_sun=300, unit_earth=301

  ! Initialize bodies
  sun%mass = 1000.0
  sun%pos = [0.0, 0.0]
  sun%vel = [0.0, 0.0]

  earth%mass = 1.0
  earth%pos = [1.0, 0.0]
  earth%vel = [0.0, 6.28]  ! roughly circular orbit

  dt = 0.01
  nsteps = 10000

  OPEN(unit=unit_sun, file='sun.dat')
  OPEN(unit=unit_earth, file='earth.dat')

  DO i = 1, nsteps
    CALL compute_force(earth, sun, fx, fy)
    CALL update_body(earth, fx, fy, dt)
    CALL update_body(sun, -fx, -fy, dt)  ! apply equal and opposite force

    IF (MOD(i, 10) == 0) THEN
      CALL sun%print(i,unit_sun)
      CALL earth%print(i,unit_earth)
    END IF
  END DO

END PROGRAM two_body_sim
```
