It'd be nice to have a super-iper-mega code that summarises all the lectures:

- For Loops
- If/elseif/else
- functions
- Static arrays
- Pointers (malloc, free, arithmetics)
- argc/argv
- Print
- Strings

```C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

// Function declaration
int factorial(int n);
void fill_array(double *arr, int size);
void print_array(const double *arr, int size);

int main(int argc, char *argv[]) {
    // 1. Print program name and arguments
    printf("Program name: %s\n", argv[0]);
    printf("You have entered %d arguments:\n", argc);
    for (int i = 1; i < argc; i++) {
        printf("arg[%d] = %s\n", i, argv[i]);
    }

    // 2. Use if/else and string comparison
    if (argc > 1 && strcmp(argv[1], "hello") == 0) {
        printf("Hello to you too!\n");
    } else if (argc > 1) {
        printf("First argument is not 'hello'.\n");
    } else {
        printf("No argument provided.\n");
    }

    // 3. Use for loop and a function (factorial)
    printf("\nFactorials from 1 to 5:\n");
    for (int i = 1; i <= 5; i++) {
        printf("%d! = %d\n", i, factorial(i));
    }

    // 4. Static array and fill
    double static_array[5];
    fill_array(static_array, 5);
    printf("\nContents of static array:\n");
    print_array(static_array, 5);

    // 5. Pointers: malloc, fill, access, free
    int n = 10;
    double *dynamic_array = (double *)malloc(n * sizeof(double));
    if (dynamic_array == NULL) {
        fprintf(stderr, "Memory allocation failed!\n");
        return 1;
    }

    fill_array(dynamic_array, n);
    printf("\nContents of dynamically allocated array:\n");
    print_array(dynamic_array, n);

    // Pointer arithmetic
    printf("\nPointer arithmetic example:\n");
    printf("Dyncamic arrays address: %p\n", dynamic_array);
    for (int i = 0; i < n; i++) {
        printf("%d) Address: %p; Value: %.2f\n", i, dynamic_array+i, *(dynamic_array + i));
    }

    free(dynamic_array);  // 6. Free memory

    return 0;
}

// Compute factorial of an integer
int factorial(int n) {
    if (n == 0 || n == 1) return 1;
    return n * factorial(n - 1);
}

// Fill an array with values i^2 + 0.5
void fill_array(double *arr, int size) {
    for (int i = 0; i < size; i++) {
        arr[i] = i * i + 0.5;
    }
}

// Print the content of an array
void print_array(const double *arr, int size) {
    for (int i = 0; i < size; i++) {
        printf("arr[%d] = %.2f\n", i, arr[i]);
    }
}
```
